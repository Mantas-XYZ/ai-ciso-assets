# Canonical source for the render workflow that lives in the PUBLIC repo at
# Mantas-XYZ/ai-ciso-assets -> .github/workflows/render-card.yml
#
# It is kept here because the workflow is part of this pipeline and belongs
# with the rest of it. Creating or editing a file under .github/workflows/
# requires the `workflow` OAuth scope, which the CLI token does not carry, so
# the public copy is maintained through the GitHub web editor.
#
# If you change this file, paste it into the web editor for the public repo.

name: Render card

# Triggered by Cowork via repository_dispatch. The card spec travels in the
# payload rather than being committed anywhere — Notion stays the source of
# truth and this repo stays a pure asset host.
#
# Deliberately NOT triggered by pull_request: this is a public repo, and a
# fork PR must never get a shot at the deploy key. Fork PRs do not receive
# secrets, but not offering the trigger at all is the stronger guarantee.
on:
  repository_dispatch:
    types: [render-card]
  workflow_dispatch:
    inputs:
      name:
        description: "Output filename, without .gif (e.g. daily-2026-08-09)"
        required: true
      spec:
        description: "Card spec as JSON"
        required: true

permissions:
  contents: write

concurrency:
  group: render-card
  cancel-in-progress: false

jobs:
  render:
    runs-on: ubuntu-latest
    steps:
      - name: Check out asset repo
        uses: actions/checkout@v4

      - name: Check out private render toolkit
        uses: actions/checkout@v4
        with:
          repository: Mantas-XYZ/ai-ciso-visuals
          ssh-key: ${{ secrets.TOOLKIT_DEPLOY_KEY }}
          path: toolkit

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - run: pip install --quiet Pillow

      - name: Resolve inputs
        id: inputs
        env:
          DISPATCH_NAME: ${{ github.event.client_payload.name }}
          DISPATCH_SPEC: ${{ toJSON(github.event.client_payload.spec) }}
          MANUAL_NAME: ${{ github.event.inputs.name }}
          MANUAL_SPEC: ${{ github.event.inputs.spec }}
        run: |
          set -euo pipefail
          if [ "${{ github.event_name }}" = "repository_dispatch" ]; then
            NAME="$DISPATCH_NAME"
            printf '%s' "$DISPATCH_SPEC" > spec.json
          else
            NAME="$MANUAL_NAME"
            printf '%s' "$MANUAL_SPEC" > spec.json
          fi
          # filenames become public URLs — keep them boring and predictable
          if ! printf '%s' "$NAME" | grep -Eq '^[a-z0-9][a-z0-9._-]{0,80}$'; then
            echo "::error::invalid name '$NAME' (lowercase letters, digits, . _ - only)"
            exit 1
          fi
          python -c "import json,sys; json.load(open('spec.json'))" \
            || { echo "::error::spec is not valid JSON"; exit 1; }
          echo "name=$NAME" >> "$GITHUB_OUTPUT"
          echo "year=$(date -u +%Y)" >> "$GITHUB_OUTPUT"

      - name: Render
        run: |
          set -euo pipefail
          mkdir -p "gifs/${{ steps.inputs.outputs.year }}"
          python toolkit/cards/render_from_spec.py \
            --spec spec.json \
            --out "gifs/${{ steps.inputs.outputs.year }}/${{ steps.inputs.outputs.name }}.gif"

      - name: Commit
        id: commit
        run: |
          set -euo pipefail
          REL="gifs/${{ steps.inputs.outputs.year }}/${{ steps.inputs.outputs.name }}.gif"
          rm -f spec.json
          git config user.name  "ai-ciso-bot"
          git config user.email "actions@github.com"
          git add "$REL"
          if git diff --cached --quiet; then
            echo "no change — asset is byte-identical to what is already published"
          else
            git commit -q -m "Render ${{ steps.inputs.outputs.name }}"
            git push -q
          fi
          echo "url=https://raw.githubusercontent.com/${{ github.repository }}/main/$REL" >> "$GITHUB_OUTPUT"

      - name: Publish URL
        run: |
          echo "### Rendered" >> "$GITHUB_STEP_SUMMARY"
          echo '```' >> "$GITHUB_STEP_SUMMARY"
          echo "${{ steps.commit.outputs.url }}" >> "$GITHUB_STEP_SUMMARY"
          echo '```' >> "$GITHUB_STEP_SUMMARY"
